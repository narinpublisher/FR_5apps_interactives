// Compteur global — incrémenté à chaque nouvelle dépense
let prochainId = 1;

function creerDepense(intitule, montant, categorie) {
  const nouvelleDepense = {
    id        : prochainId,
    intitule  : intitule,
    montant   : montant,
    categorie : categorie
  };
  prochainId++; // Incrémenter pour la prochaine dépense
  return nouvelleDepense;
}

// Exemple d'utilisation :
const d1 = creerDepense('Loyer', 850, 'Logement');
console.log(d1); // { id: 1, intitule: 'Loyer', montant: 850, categorie: 'Logement' }

function rendreInterface() {
  const liste = document.querySelector('#liste-depenses');
  const spanTotal = document.querySelector('#total');

  // Effacer la liste actuelle
  liste.innerHTML = '';

  // Obtenir les dépenses à afficher (avec filtre éventuel)
  const depensesVisibles = obtenirDepensesFiltrees();

  // Construire un <li> pour chaque dépense visible
  depensesVisibles.forEach(depense => {
    const li = document.createElement('li');
    li.className = 'item-depense';
    li.dataset.id = depense.id;

    li.innerHTML = `
      <div class="info-depense">
        <span class="intitule">${depense.intitule}</span>
        <span class="badge badge-${depense.categorie.toLowerCase()}">
          ${depense.categorie}
        </span>
      </div>
      <div class="droite">
        <span class="montant">${depense.montant.toFixed(2)} €</span>
        <button class="btn-suppr" data-id="${depense.id}">✕</button>
      </div>
    `;
    liste.appendChild(li);
  });

  // Mettre à jour le total
  const total = calculerTotal(depenses);
  spanTotal.textContent = total.toFixed(2) + ' €';

  // Mettre à jour la répartition
  afficherRepartition();
}

🎯 4.5 — Calcul des totaux avec reduce() : explication pas à pas, totaux cumulatifs

👉 Comprendre reduce() par l'exemple
--------------------
// Tableau de nombres simples
const nombres = [10, 20, 30, 40];

// Avec une boucle classique :
let total = 0;
for (let n of nombres) {
  total += n;
}
console.log(total); // 100

// Avec reduce() :
const total2 = nombres.reduce((accumulateur, valeurCourante) => {
  return accumulateur + valeurCourante;
}, 0); // 0 est la valeur de départ de l'accumulateur

console.log(total2); // 100 — même résultat, code plus court
--------------------

👉 Déroulement pas à pas
--------------------
// nombres = [10, 20, 30, 40], valeur initiale = 0

// Itération 1 : accumulateur = 0,  valeurCourante = 10  → renvoie 10
// Itération 2 : accumulateur = 10, valeurCourante = 20  → renvoie 30
// Itération 3 : accumulateur = 30, valeurCourante = 30  → renvoie 60
// Itération 4 : accumulateur = 60, valeurCourante = 40  → renvoie 100

// Résultat final : 100
--------------------

👉 Appliquer reduce() à nos dépenses
--------------------
function calculerTotal(tableau) {
  return tableau.reduce((acc, depense) => {
    return acc + depense.montant; // On accumule les montants
  }, 0); // Valeur initiale : 0
}

// Exemple avec 3 dépenses :
const mesDepenses = [
  { id: 1, intitule: 'Loyer',    montant: 850,  categorie: 'Logement'     },
  { id: 2, intitule: 'Courses',  montant: 67.40, categorie: 'Alimentation' },
  { id: 3, intitule: 'Cinéma',   montant: 12,   categorie: 'Loisirs'      },
];

console.log(calculerTotal(mesDepenses)); // 929.4
console.log(calculerTotal(mesDepenses).toFixed(2)); // '929.40'

function calculerTotauxParCategorie() {
  return depenses.reduce((acc, depense) => {
    const cat = depense.categorie;

    // Si la catégorie n'existe pas encore dans l'accumulateur, l'initialiser à 0
    if (!acc[cat]) {
      acc[cat] = 0;
    }

    // Ajouter le montant
    acc[cat] += depense.montant;

    return acc;
  }, {}); // Valeur initiale : objet vide
}

🎯  4.7 — Affichage par catégorie : patron group-by, barres de pourcentage en CSS

👉 Le patron group-by
--------------------
function calculerRepartition() {
  if (depenses.length === 0) return {};

  const totalGeneral = calculerTotal(depenses);

  // Étape 1 : grouper et sommer par catégorie
  const parCategorie = depenses.reduce((acc, d) => {
    if (!acc[d.categorie]) acc[d.categorie] = 0;
    acc[d.categorie] += d.montant;
    return acc;
  }, {});

  // Étape 2 : convertir en tableau avec pourcentages
  const resultat = {};
  for (const cat in parCategorie) {
    resultat[cat] = {
      total      : parCategorie[cat],
      pourcentage: (parCategorie[cat] / totalGeneral * 100).toFixed(1)
    };
  }
  return resultat;
}

function afficherRepartition() {
  const zone = document.querySelector('#repartition');
  const repartition = calculerRepartition();

  if (Object.keys(repartition).length === 0) {
    zone.innerHTML = '';
    return;
  }

  let html = '<h2>Répartition par catégorie</h2>';

  for (const cat in repartition) {
    const { total, pourcentage } = repartition[cat];
    const couleur = COULEURS_CAT[cat] || '#999';

    html += `
      <div class="ligne-categorie">
        <div class="label-cat">
          <span class="nom-cat">${cat}</span>
          <span class="montant-cat">${total.toFixed(2)} €
            (${pourcentage} %)</span>
        </div>
        <div class="barre-fond">
          <div class="barre-remplie"
               style="width: ${pourcentage}%;
                      background-color: ${couleur};">
          </div>
        </div>
      </div>
    `;
  }

  zone.innerHTML = html;
}

  spanTotal.textContent = calculerTotal(depenses).toFixed(2) + ' €';

  afficherRepartition();
}

function afficherRepartition() {
  const repartition = calculerRepartition();
  if (Object.keys(repartition).length === 0) {
    zoneRepartition.innerHTML = '';
    return;
  }
  let html = '<h2>Répartition par catégorie</h2>';
  for (const cat in repartition) {
    const { total, pourcentage } = repartition[cat];
    const couleur = COULEURS_CAT[cat] || '#999';
    html += `
      <div class="ligne-categorie">
        <div class="label-cat">
          <span class="nom-cat">${cat}</span>
          <span class="montant-cat">${total.toFixed(2)} € (${pourcentage} %)</span>
        </div>
        <div class="barre-fond">
          <div class="barre-remplie"
               style="width:${pourcentage}%; background:${couleur}">
          </div>
        </div>
      </div>
    `;
  }
  zoneRepartition.innerHTML = html;
}
