🌳 3.3 — Conversion chaîne vers nombre


🌸 Le problème
-----------------------------
// Ce que vous pensez avoir :
const saisie = inputMontant.value; // L'utilisateur a tapé "48.50"
 
// Ce que vous avez réellement :
console.log(typeof saisie); // "string"  ← pas un nombre !
 
// Le danger de la concaténation au lieu de l'addition :
console.log(saisie + 10); // "48.5010"  ← concatène, n'additionne pas !
-----------------------------


🌸 Les trois solutions
-----------------------------
const saisie = "48.50";
 
// 1. parseInt() — entiers seulement, ignore ce qui suit
console.log(parseInt(saisie));     // 48  (perd les décimales)
console.log(parseInt("12abc"));    // 12  (ignore "abc")
 
// 2. parseFloat() — décimaux, ignore ce qui suit
console.log(parseFloat(saisie));   // 48.5
console.log(parseFloat("12.5kg")); // 12.5  (ignore "kg")
 
// 3. Number() — conversion stricte
console.log(Number(saisie));       // 48.5
console.log(Number("12abc"));      // NaN  (Number est strict)
console.log(Number(""));           // 0
-----------------------------


🌸 NaN — le piège à connaître
-----------------------------
console.log(Number("bonjour")); // NaN
 
// ATTENTION : NaN n'est pas égal à NaN !
console.log(NaN === NaN); // false
 
// Pour détecter NaN, utilisez toujours isNaN()
console.log(isNaN(NaN));         // true
console.log(isNaN(42));          // false
console.log(isNaN("bonjour"));   // true
 
// Dans notre application, on vérifiera :
if (isNaN(montant) || montant <= 0) {
  // Ne pas calculer, afficher une erreur
}

🌳 3.4 — Calcul et arrondi

🌸 La fonction calculer()
-----------------------------
function calculer() {
  // 1. Lire et convertir les saisies
  const montant   = parseFloat(inputMontant.value);
  const personnes = parseInt(inputPersonnes.value);
 
  // 2. Validation : arrêter si les valeurs sont invalides
  if (isNaN(montant) || montant <= 0 ||
      isNaN(personnes) || personnes < 1) {
    spanPourboirePP.textContent = '0.00 €';
    spanTotalPP.textContent     = '0.00 €';
    return; // Sortir de la fonction
  }
 
  // 3. Calcul du pourboire total
  const pourboire = montant * (pourcentageTip / 100);
 
  // 4. Total addition + pourboire
  const total = montant + pourboire;
 
  // 5. Division par personne
  const pourboireParPersonne = pourboire / personnes;
  const totalParPersonne     = total / personnes;
 
  // 6. Afficher les résultats (voir section 3.5)
  spanPourboirePP.textContent =
    pourboireParPersonne.toFixed(2) + ' €';
  spanTotalPP.textContent =
    totalParPersonne.toFixed(2) + ' €';
}
-----------------------------


🌸 Math.round et toFixed()
-----------------------------
// Math.round() — arrondit à l'entier le plus proche
console.log(Math.round(12.6));  // 13
console.log(Math.round(12.4));  // 12
 
// toFixed(n) — arrondit ET formate avec n décimales
// Attention : toFixed() renvoie une chaîne, pas un nombre !
console.log((12.3456).toFixed(2));  // "12.35"
console.log((10).toFixed(2));       // "10.00"
 
// Dans notre projet, toFixed(2) est parfait :
// - Affiche toujours 2 décimales (format montant)
// - Arrondit correctement (0.005 → arrondi supérieur)
const resultat = (48.5 * 0.15 / 4).toFixed(2);
console.log(resultat); // "1.82"

🌳 3.5 — Mise à jour du DOM

🌸 textContent vs innerHTML
-----------------------------
// textContent — texte pur, sécurisé (recommandé)
spanPourboirePP.textContent = '2.25 €';
// Affiche littéralement : 2.25 €
 
// innerHTML — interprète le HTML, moins sécurisé
spanPourboirePP.innerHTML = '<strong>2.25 €</strong>';
// Affiche : 2.25 € en gras
 
// DANGER avec innerHTML et données utilisateur :
const nom = "<img src=x onerror='alert(1)'>";
element.innerHTML = nom; // ❌ Risque de sécurité (XSS)
element.textContent = nom; // ✅ Affiche le texte brut, sans interpréter

🌳 3.6 — Gestion des erreurs & validation

🌸 La validation complète
-----------------------------
function calculer() {
  const montant   = parseFloat(inputMontant.value);
  const personnes = parseInt(inputPersonnes.value);
 
  // Réinitialiser l'affichage si les données sont invalides
  if (isNaN(montant) || montant <= 0 ||
      isNaN(personnes) || personnes < 1) {
    spanPourboirePP.textContent = '0.00 €';
    spanTotalPP.textContent     = '0.00 €';
    afficherErreur(!isNaN(montant) || montant > 0);
    return;
  }
 
  // Masquer le message d'erreur si tout est bon
  masquerErreur();
 
  // ... (suite du calcul)
}
 
// Afficher un message d'erreur dans la page
function afficherErreur(champValide) {
  const zone = document.querySelector('#message-erreur');
  zone.textContent = champValide
    ? 'Veuillez entrer un nombre de personnes valide (minimum 1).'
    : 'Veuillez entrer un montant valide et positif.';
  zone.style.display = 'block';
}
 
function masquerErreur() {
  document.querySelector('#message-erreur').style.display = 'none';
}

function calculer() {
  const montant   = parseFloat(inputMontant.value);
  const personnes = parseInt(inputPersonnes.value);
 
  if (isNaN(montant) || montant <= 0 ||
      isNaN(personnes) || personnes < 1) {
    spanPourboirePP.textContent = '0.00 €';

    (pourboire / personnes).toFixed(2) + ' €';

    (total / personnes).toFixed(2) + ' €';
