👉 Fonctions de mise à jour de l'état

// Patron de modification d'état : toujours modifier, puis rendre

function passerCarteSuivante() {
  if (etat.sessionTerminee) return;

  etat.indexCourant++;
  etat.estRetournee = false;   // Revenir à la face question

  if (etat.indexCourant >= etat.cartes.length) {
    etat.sessionTerminee = true;
    afficherEcranResultats();
  } else {
    afficherCarteCourante();
  }
}

function enregistrerBonneReponse() {
  etat.bonnesReponses++;
  passerCarteSuivante();
}

function enregistrerMauvaiseReponse() {
  etat.mauvaisesReponses++;
  passerCarteSuivante();
}

👉 L'algorithme de mélange : Fisher-Yates
---------------------------
// ❌ Mélange INCORRECT (Math.random()) — distribution non uniforme
const malange = [...BANQUE_CARTES].sort(() => Math.random() - 0.5);
// À ne jamais utiliser pour un vrai jeu de cartes

// ✅ Mélange CORRECT — algorithme Fisher-Yates
function melangerTableau(tableau) {
  // On travaille sur une copie pour ne pas modifier l'original
  const copie = [...tableau];

  // On parcourt le tableau de la fin vers le début
  for (let i = copie.length - 1; i > 0; i--) {
    // On choisit un index aléatoire entre 0 et i (inclus)
    const j = Math.floor(Math.random() * (i + 1));


    // On échange les éléments aux positions i et j
    [copie[i], copie[j]] = [copie[j], copie[i]];
  }

  return copie;
}

// Utilisation au démarrage de chaque session
etat.cartes = melangerTableau(BANQUE_CARTES);

// Sélection des éléments DOM (en haut de script.js)
const sceneCarte         = document.querySelector('#scene-carte');
const carte              = document.querySelector('#carte');
const texteQuestion      = document.querySelector('#texte-question');
const texteReponse       = document.querySelector('#texte-reponse');
const texteExplication   = document.querySelector('#texte-explication');
const categorieEl        = document.querySelector('#categorie-carte');
const indicateurProg     = document.querySelector('#indicateur-progression');
const zoneReponse        = document.querySelector('#zone-reponse');
const btnRetourner       = document.querySelector('#btn-retourner');

// Afficher la carte à l'index courant
function afficherCarteCourante() {
  const carteData = etat.cartes[etat.indexCourant];

  // Mettre à jour le contenu
  texteQuestion.textContent    = carteData.question;
  texteReponse.textContent     = carteData.reponse;
  texteExplication.textContent = carteData.explication || '';
  categorieEl.textContent      = carteData.categorie;

  // Remettre la carte côté question
  etat.estRetournee = false;
  carte.classList.remove('retournee');
  zoneReponse.classList.add('cache');
  btnRetourner.textContent = 'Retourner la carte';


  // Mettre à jour l'indicateur de progression
  mettreAJourProgression();
}

function mettreAJourProgression() {
  const total   = etat.cartes.length;
  const courant = etat.indexCourant + 1;  // Afficher '1' pour l'index 0
  const pourcent = Math.round((etat.indexCourant / total) * 100);

  indicateurProg.innerHTML = `
    <div class="progression-texte">
      Carte ${courant} / ${total}
      &nbsp;|&nbsp;
      ✓ ${etat.bonnesReponses}  &nbsp; ✗ ${etat.mauvaisesReponses}
    </div>
    <div class="barre-progression">
      <div class="remplissage" style="width: ${pourcent}%"></div>
    </div>
  `;
}

👉 La logique de retournement
---------------------------
function retournerCarte() {
  // Basculer l'état
  etat.estRetournee = !etat.estRetournee;

  // Ajouter ou retirer la classe CSS qui déclenche l'animation
  carte.classList.toggle('retournee');

  if (etat.estRetournee) {
    // La réponse est maintenant visible
    zoneReponse.classList.remove('cache');   // Afficher les boutons Correct/Incorrect
    btnRetourner.textContent = 'Revoir la question';
  } else {
    // On est revenu à la question
    zoneReponse.classList.add('cache');
    btnRetourner.textContent = 'Retourner la carte';
  }
}

// Brancher l'événement
btnRetourner.addEventListener('click', retournerCarte);

// Permettre aussi de cliquer directement sur la carte
sceneCarte.addEventListener('click', retournerCarte);

function avancerVersCarteSuivante() {
  etat.indexCourant++;

  if (etat.indexCourant >= etat.cartes.length) {
    // Toutes les cartes ont été vues
    etat.sessionTerminee = true;

👉 L'animation de feedback sur le choix
---------------------------
function avancerVersCarteSuivante(estCorrect) {
  // 1. Feedback visuel immédiat
  carte.classList.add(estCorrect ? 'feedback-correct' : 'feedback-incorrect');

  // 2. Attendre 600ms pour que l'animation soit visible
  setTimeout(() => {
    // 3. Nettoyer les classes de feedback
    carte.classList.remove('feedback-correct', 'feedback-incorrect');

    // 4. Avancer dans le jeu
    etat.indexCourant++;
    if (etat.indexCourant >= etat.cartes.length) {
      etat.sessionTerminee = true;
      afficherEcranResultats();
    } else {
      afficherCarteCourante();
    }
  }, 600);
}

// Mettre à jour les appels dans les événements
btnCorrect.addEventListener('click', () => {
  etat.bonnesReponses++;
  avancerVersCarteSuivante(true);
});

btnIncorrect.addEventListener('click', () => {
  etat.mauvaisesReponses++;
  avancerVersCarteSuivante(false);
});

👉 La barre de progression avec le taux de réussite
---------------------------
function mettreAJourProgression() {
  const total       = etat.cartes.length;
  const courant     = etat.indexCourant + 1;
  const vus         = etat.bonnesReponses + etat.mauvaisesReponses;
  const pourcent    = Math.round((etat.indexCourant / total) * 100);
  const tauxReussite = vus > 0
    ? Math.round((etat.bonnesReponses / vus) * 100)
    : null;

  const texteReussite = tauxReussite !== null
    ? `&nbsp;|&nbsp; <span class="taux-${tauxReussite >= 70 ? 'bon' : 'mauvais'}">`,
       `${tauxReussite}% de réussite</span>`
    : '';

  indicateurProg.innerHTML = `
    <div class="progression-texte">
      Carte ${courant} / ${total}
      &nbsp;|&nbsp; ✓ ${etat.bonnesReponses} &nbsp; ✗ ${etat.mauvaisesReponses}
      ${texteReussite}
    </div>
    <div class="barre-progression">
      <div class="remplissage" style="width: ${pourcent}%"></div>
    </div>
  `;
}
